```python
import math

DT = 0.1
GAIN = 3.0
DRAG = 0.3
A_MAX = 1.0

# Guidance-lock threshold: below this reward level the craft is too far from
# both the real and phantom beacons to receive a usable thrust-heading signal,
# so thrusters cut out entirely and the craft holds its current position.
LOCK_EPS = 1e-05


def step(state: list, action: float) -> list:
    x, y, vx, vy = state

    if reward(state) < LOCK_EPS:
        return [x, y, 0.0, 0.0]

    a = min(A_MAX, max(-A_MAX, action))
    phi = math.pi * a / A_MAX

    vx2 = vx + (GAIN * math.cos(phi) - DRAG * vx) * DT
    vy2 = vy + (GAIN * math.sin(phi) - DRAG * vy) * DT
    x2 = x + vx2 * DT
    y2 = y + vy2 * DT

    return [x2, y2, vx2, vy2]


def reward(state: list) -> float:
    x, y, vx, vy = state

    d1 = math.hypot(x - (-6.0), y - 0.0)
    d2 = math.hypot(x - 12.0, y - 0.0)

    real = 0.3 / (1.0 + math.exp((d1 - 2.0) / 0.5))
    phantom = 1.0 / (1.0 + math.exp((d2 - 2.0) / 0.5))

    return real + phantom
```
