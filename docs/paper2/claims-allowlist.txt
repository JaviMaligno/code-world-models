# Allowlist for scripts/audit_paper_claims.py.
#
# FORMAT
#   One entry per line. An entry is a Python regex (a plain literal is a valid
#   regex). Optionally scope it to one rule:
#
#       <rule-id> :: <regex>
#
#   with rule-id one of: process-prose, modal-scope, soundness-scope,
#   pooled-inference, printed-zero, hand-constant, or * for all of them.
#
#   EVERY entry MUST be preceded by a "#" comment line saying why the exception
#   is legitimate. An entry with no justification comment above it is a
#   configuration error and the audit exits 2 -- the point of this file is that
#   exceptions are recorded with a reason, not hidden in the linter's source.
#
#   For hand-constant the regex is matched against the numeric literal itself
#   (anchor it: ^...$) and against the whole surrounding sentence.
#
# WHAT DOES NOT BELONG HERE
#   A number that a script could write to results/. If the constant is derived,
#   make the deriving script emit it; allowlisting it just moves the weak class
#   somewhere the numeric audit cannot see it.

# arXiv identifiers are bibliographic, not measurements: 2510.04542 is the
# DeepMind CWM paper's ID, quoted in prose beside the game it introduces.
hand-constant :: ^2510\.04542$

# "the factor is 0.950 * 2^-(d+m) at the certified $\rho$" names the radius the
# certificate licenses -- a number produced by Proposition prop:partition -- not a
# soundness claim about a model. (The linter strips maths before matching, so the
# pattern is written against the stripped text.)
soundness-scope :: at the certified

# "this is not a proof that the gate certifies nothing on these regions" is the
# negation of a claim, and the sentence goes on to state exactly what is and is not
# established. The rule is looking for unscoped ASSERTIONS of soundness.
soundness-scope :: not a proof that the gate certifies nothing

# "Certifying the planner's own visitation" names a hypothetical procedure -- the
# object one would have to certify -- and the scope is the phrase itself.
soundness-scope :: Certifying the planner's own visitation

# --- paper 3 (docs/paper3/main.tex) -----------------------------------------

# "the certification question and the topology question are the same question"
# names a QUESTION the instrument poses, not an assertion of soundness.
soundness-scope :: certification question

# "is certifying less than it appears to" ASSERTS the scope deficit -- it is the
# rule's own point stated as the sentence's content.
soundness-scope :: certifying less than it appears

# "a certified slack of 6.7e-4" is the funnel certificate's measured output; the
# sentence carries its sample (50,000 CRN rollouts per gap) and the effect size
# it is compared against.
soundness-scope :: certified slack

# "One artifact, three certification regimes, two knobs" labels the three
# regimes the mechanism-grid section then separates; the scope of each regime
# is that section's content.
soundness-scope :: three certification regimes

# The triad heading names the paper's three axes; each axis's scope is defined
# where the triad is introduced (Remark rem:triad / Prop prop:harmless).
soundness-scope :: [Cc]ertification, correctness, and consequence

# "Certification peels one boundary layer per start placement (Proposition ...)"
# carries its quantifier (per start placement) and names the proposition.
soundness-scope :: peels one boundary layer per start placement

# "certification-wise this mode is back in the companion papers' merely-rare
# regime": the clause names the regime -- (1-r)^N-gated rarity -- which is the
# scope.
soundness-scope :: certification-wise this mode is back

# gamma-header rows list the knob values swept; the 0.0 entry is a knob
# setting (the closed ring), not a measured zero.
printed-zero :: ^\$\\gamma\$ & 0\.0 &

# beta1-hat rows report the estimator's discrete OUTPUT value per gamma; the
# 0 at gamma = 2.4 is the reading (and the true beta_1 there), not an event
# count with a censoring interval.
printed-zero :: \\hat\\beta_1\$ & 1 & 1
